static String reverseWords(String s) {
    String[] parts = s.trim().split("\\s+");
    StringBuilder sb = new StringBuilder();
    for (int i = parts.length - 1; i > 0; i--) {
        sb.append(parts[i]);
        if (i > 0) sb.append(" ");
    }
    return sb.toString();
}

--- code
static boolean hasPairWithSum(int[] sorted, int target) {
    int lo = 0, hi = sorted.length - 1;
    while (lo <= hi) {
        int s = sorted[lo] + sorted[hi];
        if (s == target) return true;
        if (s < target) lo++;
        else hi--;
    }
    return false;
}

goal: Shift each lowercase letter forward by k positions (0 <= k < 26), wrapping around; keep other characters.
bugLine: 5
explanation: char plus int is an int, so the character code number is appended instead of a letter.
fixed:             sb.append((char) ('a' + (c - 'a' + k) % 26));
--- code
static String caesar(String s, int k) {
    StringBuilder sb = new StringBuilder();
    for (char c : s.toCharArray()) {
        if (c >= 'a' && c <= 'z') {
            sb.append('a' + (c - 'a' + k) % 26);
        } else {
            sb.append(c);
        }
    }
    return sb.toString();
}

=== java-h3
language: java
difficulty: hard
goal: Return the most frequent letter in a non-empty lowercase string, picking the earliest letter on ties.
bugLine: 8
explanation: >= lets a later letter with an equal count replace the earlier one.
fixed:         if (counts[i] > counts[best]) best = i;
--- code
static char mostFrequent(String s) {
    int[] counts = new int[26];
    for (char c : s.toCharArray()) {
        counts[c - 'a']++;
    }
    int best = 0;
    for (int i = 1; i < 26; i++) {
        if (counts[i] >= counts[best]) best = i;
    }
    return (char) ('a' + best);
}

static boolean sameCounts(Map<String, Integer> a, Map<String, Integer> b) {
    if (a.size() != b.size()) {
        return false;
    }
    for (String key : a.keySet()) {
        if (a.get(key) != b.get(key)) {
            return false;
        }
    }
    return true;
}

static int[] bucketSizes(String[] keys, int buckets) {
    int[] sizes = new int[buckets];
    for (String key : keys) {
        int hash = key.hashCode();
        int idx = Math.abs(hash) % buckets;
        sizes[idx]++;
    }
    return sizes;
}
