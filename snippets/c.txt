=== c-m1
language: c
difficulty: medium
goal: Reverse a NUL-terminated string in place.
bugLine: 2
explanation: j starts on the terminating NUL, which gets swapped to the front and empties the string.
fixed:     int j = (int)strlen(s) - 1;
--- code
void reverse(char *s) {
    int j = (int)strlen(s);
    int i = 0;
    while (i < j) {
        char tmp = s[i];
        s[i] = s[j];
        s[j] = tmp;
        i++;
        j--;
    }
}

=== c-m3

goal: Parse an optional '-' followed by decimal digits into an int (stops at the first non-digit).
bugLine: 9
explanation: It adds the character code instead of the digit value.
fixed:         value = value * 10 + (*s - '0');
--- code
int parse_int(const char *s) {
    int sign = 1;
    int value = 0;
    if (*s == '-') {
        sign = -1;
        s++;
    }
    while (*s >= '0' && *s <= '9') {
        value = value * 10 + *s;
        s++;
    }
    return sign * value;
}

=== c-x3

bugLine: 4
explanation: The buffer has no room for the space, so the final NUL is written past the end.
fixed:     char *out = malloc(la + lb + 2);
--- code
char *join_words(const char *a, const char *b) {
    size_t la = strlen(a);
    size_t lb = strlen(b);
    char *out = malloc(la + lb + 1);
    if (out == NULL) {
        return NULL;
    }
    memcpy(out, a, la);
    out[la] = ' ';
    memcpy(out + la + 1, b, lb + 1);
    return out;
}
